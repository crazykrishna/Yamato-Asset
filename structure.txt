1. LOGIN.HTML -

2. HOME.HTML -
	- WELCOME {USERNAME}
	- BASIC INFORMATION
	- GRAPHS OF -
		- TOTAL CATEGORY
		- INVENTORY

3. ASSETS.HTML-
	- TABLE OF ALL ASSETS
	- CONTENTS OF TABLE :-
		- HOST_ID

		- EMPLOYEE_NUMBER
		- EMPLOYEE_EMAIL
		- MODEL_NUMBER
		- ASSET_PURCHASE_DATE
		- REMARKS (IF ANY)

	- SEARCH AND FILTER BAR
		
4. INVENTORY.HTML -
	- TABLE OF INVENTRY ASSETS --- CONTENT --->
		- ASSET_ID
		- ASSET_NAME
		- ASSET_MODEL
		- ASSET_PURCHASE_DATE
		- ASSET_LICENSE_DATE
		- ASSET_TYPE

	- SEARCH AND FILTER BAR

5. REPORTS.HTML -
	- REPORT OF SPECIFIC OR ALL EMPLOYEE(S)

6. REGISTER.HTML - 
	- FORM -
		- NAME
		- EMPLOYEE_ID
		- EMAIL
		- PHONE NUMBER
		- DEPARTMENT
		- TYPE OF ASSET REQUIRED

7. ASSET REGISTER.HTML -
	FORM - 
		- ASSET_ID
		- ASSET_NAME
		- ASSET_MODEL
